# Taming and Release 4Ever By ZentosPlay - https://www.youtube.com/@ZentosPlay
# RELEASE + RE-TAMING FOREVER (DUAL PET VERSION)
# Plays until you manually pause it
# Releases pet 1, releases pet 2, tames pet 1, tames pet 2, repeat.
########################################
# V 11.0  - Last Updated: 03/06/2026 ##
########################################
@setvar! tameDelay 4500
@setvar! releaseDelay 500
########################################
### SELECT PET 1
########################################
overhead 'SELECT PET 1 - Click your first pet now!' 88
sysmsg 'SELECT PET 1: Click your FIRST pet.'
wait 500
setvar 'trainPet1'
wait 500
overhead 'PET 1 SELECTED! Get ready for Pet 2...' 68
wait 2000
########################################
### SELECT PET 2
########################################
overhead 'SELECT PET 2 - Click your second pet now!' 88
sysmsg 'SELECT PET 2: Click your SECOND pet.'
wait 500
setvar 'trainPet2'
wait 500
overhead 'PET 2 SELECTED! Starting in 3 seconds...' 68
wait 3000
########################################
### LOOP
########################################
while not dead
    ################################
    ### RELEASE PET 1
    ################################
    overhead 'Releasing Pet 1...' 88
    say 'a husk crab release'
    wait 250
    waitforgump 2426193729 3000
    gumpresponse 1
    wait releaseDelay

    ################################
    ### RELEASE PET 2
    ################################
    overhead 'Releasing Pet 2...' 88
    say 'TrainPet release'
    wait 250
    waitforgump 2426193729 3000
    gumpresponse 1
    wait releaseDelay

    ################################
    ### TAME PET 1
    ################################
    overhead 'Taming Pet 1...' 68
    hotkey 'Animal Taming'
    waitfortarget 1500
    target 'trainPet1'
    wait tameDelay

    ################################
    ### TAME PET 2 then rename it back
    ################################
    overhead 'Taming Pet 2...' 68
    hotkey 'Animal Taming'
    waitfortarget 1500
    target 'trainPet2'
    wait tameDelay
    rename 'trainPet2' 'TrainPet'
    wait 500

endwhile
