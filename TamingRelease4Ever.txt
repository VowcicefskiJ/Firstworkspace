# Taming and Release 4Ever By ZentosPlay - https://www.youtube.com/@ZentosPlay
# RELEASE + RE-TAMING FOREVER (DUAL PET VERSION)
# Plays until you manually pause it
# Releases pet 1, releases pet 2, tames pet 1, tames pet 2, repeat.
########################################
# V 7.0  - Last Updated: 03/06/2026 ##
########################################
@setvar! tameDelay 4500
@setvar! releaseDelay 500
########################################
### SELECT PET 1
########################################
overhead 'SELECT PET 1' 88
sysmsg 'Click on the FIRST pet you want to train.'
setvar 'trainPet1'
wait 300
########################################
### SELECT PET 2 (same type as Pet 1)
########################################
overhead 'SELECT PET 2' 88
sysmsg 'Click on the SECOND pet you want to train (same type as Pet 1).'
setvar 'trainPet2'
wait 300
########################################
### LOOP
########################################
while not dead
    ################################
    ### RELEASE PET 1
    ################################
    contextmenu 'trainPet1' 6
    wait 500
    waitforgump 0 3000
    gumpresponse 0
    wait releaseDelay

    ################################
    ### RELEASE PET 2
    ################################
    contextmenu 'trainPet2' 6
    wait 500
    waitforgump 0 3000
    gumpresponse 0
    wait releaseDelay

    ################################
    ### TAME PET 1
    ################################
    hotkey 'Animal Taming'
    waitfortarget 1500
    target 'trainPet1'
    wait tameDelay

    ################################
    ### TAME PET 2
    ################################
    hotkey 'Animal Taming'
    waitfortarget 1500
    target 'trainPet2'
    wait tameDelay

endwhile
